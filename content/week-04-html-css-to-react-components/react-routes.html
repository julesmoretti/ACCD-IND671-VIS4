Step 1: Install Redux and React-Redux

# Install Redux and React-Redux
npm install redux react-redux

To start using Redux, you need to install two libraries: redux (which is the core library) and react-redux (which helps connect Redux to your React components).

Step 2: Create and Configure Redux Store

To use Redux, you need to create a central store that holds the application's state. First, create a new file called store.js in the src/ directory:

import { createStore } from 'redux';

// Initial state - this is the main data store for your app, in this you will keep a record of anything you want to store or edit within your local app
const initialState = {
  // -----------------------------------------
  user: null,
  // This is where you would store any variable used within your app.
  // for example:
  // instagramPosts: [],
  // spotifyPlaylists: [],
  // -----------------------------------------
};

// Reducer function to handle actions
const reducer = (state = initialState, action) => {
  switch (action.type) {

    // -----------------------------------------
    // these case's are refered as Actions, and is what you need to define in order to edit the initial state
    case 'SET_USER':
      console.log('SET_USER', state, action.payload);

      return {
        ...state,
        user: action.payload
      };
    case 'CLEAR_USER':
      console.log('SET_USER', state);

      return {
        ...state,
        user: null
      };
    // end of the user defined cases, the rest is just default redux implementation.
    // -----------------------------------------

    default:
      return state;
  }
};

const store = createStore(reducer);

export default store;

Then wrap the entire application with the Provider component in index.js to make the store available to all components:

import React from 'react';
import ReactDOM from 'react-dom/client';
import './index.css';
import App from './App';
import store from './store';
import { Provider } from 'react-redux';
import reportWebVitals from './reportWebVitals';

const root = ReactDOM.createRoot(document.getElementById('root'));
root.render(
  <React.StrictMode>
    <Provider store={store}>
      <App />
    </Provider>,
  </React.StrictMode>
);

Step 3: How Redux Works at a High Level

Redux is a predictable state container for JavaScript apps. At a high level:

  • Store: A single source of truth that holds the state (list of variables within a single object) of your application.
  • Actions: Descriptions of events that change the state.
  • Reducers: Pure functions that take the current state and an action, and return the next state.
  • Dispatch: The way to trigger an action to update the state.

Step 4: Reading and Writing Redux Content in a Component

To read data from the Redux store or to dispatch actions to it, you use the useSelector and useDispatch hooks.

Example of a component reading and writting Redux content:

import React from 'react';
// the reading and writting function from redux
import { useSelector, useDispatch } from 'react-redux';

function SignOut() {
  // the useSelector in this case allows this component to read the user data from your store.js and any changes it will undergo.
  const user = useSelector((state) => state.user);
  
  // this allows this function to use dispatch which is a tool from redux that allows you to trigger a defined Action within your Store's reducer.
  const dispatch = useDispatch();
  
  const handleSignOut = () => {
    dispatch({
      // As we can see this dispatches an Action called CLEAR_USER which you have defined in your store, NO payload goes with this Action.
      type: 'CLEAR_USER'
    });
  };

  return (
    <div>
        <img src={user.photoURL.slice(0, -6)} alt="profile" />
        <p>Welcom, {user.displayName}</p>
        <button onClick={handleSignOut}>Sign Out</button>
    </div>
  );
}

export default SignOut;

Example of a component writting Redux content:

import React from 'react';
import {
  GoogleAuthProvider,
  signInWithPopup,
} from 'firebase/auth';

// the writting function from redux
import { useDispatch } from 'react-redux';
import { auth } from './firebase';

function SignIn() {  
  // this allows this function to use dispatch which is a tool from redux that allows you to trigger a defined Action within your Store's reducer.
  const dispatch = useDispatch();
  
  const handleLogin = (user) => {
  	// As we can see this dispatches an Action called SET_USER which you have defined in your store, and at the same time it is passing a payload with user
    dispatch({
      type: 'SET_USER',
      payload: user,
    });
  };
  
  const signInWithGoogle = () => {
    const provider = new GoogleAuthProvider();
    signInWithPopup(auth, provider)
      .then((result) => handleLogin(result.user))
      .catch((error) => alert(error.message));
  };

  return <button onClick={signInWithGoogle}> Sign In with Google</button>;
}

export default SignIn;

Step 5: Install and Set Up React-Router

# Install React-Router
npm install react-router-dom

To configure the app with React-Router, import the necessary components in App.js:

import { BrowserRouter as Router, Routes, Route, Link } from 'react-router-dom';
import React, { useEffect } from 'react';
import { useSelector } from 'react-redux';
import { onAuthStateChanged } from 'firebase/auth';
import { auth } from './firebase';
import './App.css';
import Profile from './Profile';
import About from './About';
import SignIn from './SignIn';
import SignOut from './SignOut';

function App() {
  // new redux method
  const user = useSelector((state) => state.user);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {});
    return () => unsubscribe();
  }, []);

  return (
  	// Router only gets defined once in the entire app
    <Router>
      <div className="App">
        <nav>
          <ul>
            <li><Link to="/">Home</Link></li>
            <li><Link to="/profile">Profile</Link></li>
            <li><Link to="/about">About</Link></li>
          </ul>
        </nav>
        <header className="App-header">
          <Routes>
            <Route path="/" element={user ? <SignOut /> : <SignIn />} />
            <Route 
              // using the /* at the end of /profile allows you to have within the profile page another set of Link's and Routes and Route
              path="/profile/*" 
              element={user ? <Profile /> : <SignIn />} 
            />
            <Route path="/about" element={user ? <About /> : <SignIn />} />
          </Routes>
        </header>
      </div>
    </Router>
  );
}

export default App;

Step 6: Creating a Profile component with a Sidebar div navigation

In the UserProfile component, you can add a left sidebar with links for different profile sections.

import React from 'react';
import { BrowserRouter as Routes, Route, Link } from 'react-router-dom';
import './Profile.css';

function Profile() {
  return (
    // No Router used as this is a secondary navigation underneath the main one.
    <div className="Profile">
      <div className="Sidebar">
        <ul>
          <li>
            <Link to="">Base</Link>
          </li>
          <li>
            <Link to="jules">Jules</Link>
          </li>
          <li>
            <Link to="lulu">Lulu</Link>
          </li>
        </ul>
      </div>
      <div className="ProfileContent">
        <Routes>
          // note that we are not using any / for the path as the App.js already has that included when using /*
          <Route path="" element=<p>Base Profile</p> />
          <Route path="jules" element=<p>Jules's Profile</p> />
          <Route path="lulu" element=<p>Lulu's Profile</p> />
        </Routes>
      </div>
    </div>
  );
}

export default Profile;
